#### Define row,cols This helps in defining boundry conditions for recursion, make visited matrix etc.
``` python
rows = len(grid)
cols = len(grid[0]) if grid else 0
```

#### Define boundry conditions
``` python
if 0 <= i < rows and 0 <= j < cols:
    print(grid[i][j])
```

#### Make visited matrix
``` python
visited=[[False]*cols for _ in range(rows)]
```

def bfs(i,j):
    queue=[(i,j)]
    while queue:
        x,y=queue.pop(0)
        if (x<0 or x>=rows or y<0 or y>=cols):
            continue
        elif visited[x][y]:
            continue
        elif grid[x][y]=="0":
            continue
        else:
            visited[x][y]=True
            print(f"Visiting cell: ({x},{y})")

def dfs(i, j):
    stack = [(i, j)]
    while stack:
        x, y = stack.pop()
        if 0 <= x < rows and 0 <= y < cols:
            if not visited[x][y] and grid[x][y] == "1":
                visited[x][y] = True
                print(f"Visiting: ({x}, {y})")
                neighbors = [(x+1, y), (x-1, y),
                             (x, y+1), (x, y-1)]
                for i, j in neighbors:
                    stack.append((i, j))
