def bfs(i,j):
    queue=[(i,j)]
    while queue:
        #  pop the first element
        x,y=queue.pop(0)
        # Check cell validity
        if (x<0 or x>=rows or y<0 or y>=cols):
            # print(f"Skipping cell: ({x},{y}) - Out of bounds")
            continue
        # check cell question condition
        elif visited[x][y]:
            # print(f"Skipping cell: ({x},{y}) - Already visited")

def dfs(graph, start):
    visited = {start}
    stack = [start]

    while stack:
        node = stack.pop()
        print(node)

        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                stack.append(neighbor)
